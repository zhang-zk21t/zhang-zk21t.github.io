{{- /* Renders GitHub-style alerts: > [!NOTE], [!TIP], [!IMPORTANT], [!WARNING], [!CAUTION]. Works whether the marker line is followed directly by content or separated by a blank > line. Falls back to a plain blockquote for non-alert quotes. */ -}} {{- $alertType := "" -}} {{- $body := .Text -}} {{- /* Case 1: Hugo's parser already detected the alert. */ -}} {{- if eq .Type "alert" -}} {{- $alertType = lower .AlertType -}} {{- else -}} {{- /* Case 2: blank-line variant — first child paragraph is just [!TYPE]. */ -}} {{- $re := `^

\s*\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]\s*

\s*` -}} {{- with findRESubmatch $re .Text 1 -}} {{- $alertType = lower (index (index . 0) 1) -}} {{- $body = replaceRE $re "" $.Text -}} {{- end -}} {{- end -}} {{- if $alertType -}} {{- $palette := dict "note" "#0969da" "tip" "#1a7f37" "important" "#8250df" "warning" "#9a6700" "caution" "#d1242f" -}} {{- $label := dict "note" "Note" "tip" "Tip" "important" "Important" "warning" "Warning" "caution" "Caution" -}} {{- $c := index $palette $alertType -}}

{{ index $label $alertType }}

{{ $body | safeHTML }}
{{- else -}}
{{ .Text | safeHTML }}
{{- end -}}
